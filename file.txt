    7. Add a Node.js and Express Backend

    The website currently uses sample restaurant data stored in React. To turn it into a full MERN project, let's create a backend API.

    Open a second terminal and navigate to the project directory:

    cd swiggy-project
    mkdir backend
    cd backend
    npm init -y
    npm install express mongoose cors dotenv

    Create the backend file:

    nano server.js

    Paste this code:

    const express = require("express");
    const mongoose = require("mongoose");
    const cors = require("cors");
    require("dotenv").config();

    const app = express();

    app.use(cors());
    app.use(express.json());

    const restaurantSchema = new mongoose.Schema({
      name: String,
      cuisine: String,
      rating: Number,
      price: Number,
    });

    const Restaurant = mongoose.model(
      "Restaurant",
      restaurantSchema
    );

    app.get("/", (req, res) => {
      res.send("Swiggy Food Delivery API is running");
    });

    app.get("/api/restaurants", async (req, res) => {
      try {
        const restaurants = await Restaurant.find();
        res.json(restaurants);
      } catch (error) {
        res.status(500).json({
          message: "Unable to fetch restaurants",
        });
      }
    });

    app.post("/api/restaurants", async (req, res) => {
      try {
        const restaurant = await Restaurant.create(req.body);
        res.status(201).json(restaurant);
      } catch (error) {
        res.status(400).json({
          message: "Unable to create restaurant",
        });
      }
    });

    const PORT = process.env.PORT || 5000;

    async function startServer() {
      try {
        if (!process.env.MONGO_URI) {
          throw new Error("MONGO_URI is missing from .env");
        }

        await mongoose.connect(process.env.MONGO_URI);

        app.listen(PORT, () => {
          console.log(`Server running on port ${PORT}`);
        });
      } catch (error) {
        console.error("Server startup failed:", error.message);
        process.exit(1);
      }
    }

    startServer();

    Save the file.
    Configure MongoDB

    Create a .env file:

    nano .env

    Add your MongoDB connection string:

    MONGO_URI=mongodb://127.0.0.1:27017/swiggy
    PORT=5000

    Save and exit.

    This connection string assumes MongoDB is installed and running locally. If you use MongoDB Atlas, replace it with your Atlas connection string.

    Start the backend:

    node server.js

    If the database connection succeeds, you should see:

    Server running on port 5000

    Open this URL in your browser to test the API:

    http://localhost:5000/

    Expected response:

    Swiggy Food Delivery API is running

    The restaurant API is available at:

    http://localhost:5000/api/restaurants

    It will return an empty array until you add restaurant records to MongoDB.

        7. Add a Node.js and Express Backend

        The website currently uses sample restaurant data stored in React. To turn it into a full MERN project, let's create a backend API.

        Open a second terminal and navigate to the project directory:

        cd swiggy-project
        mkdir backend
        cd backend
        npm init -y
        npm install express mongoose cors dotenv

        Create the backend file:

        nano server.js

        Paste this code:

        const express = require("express");
        const mongoose = require("mongoose");
        const cors = require("cors");
        require("dotenv").config();

        const app = express();

        app.use(cors());
        app.use(express.json());

        const restaurantSchema = new mongoose.Schema({
          name: String,
          cuisine: String,
          rating: Number,
          price: Number,
        });

        const Restaurant = mongoose.model(
          "Restaurant",
          restaurantSchema
        );

        app.get("/", (req, res) => {
          res.send("Swiggy Food Delivery API is running");
        });

        app.get("/api/restaurants", async (req, res) => {
          try {
            const restaurants = await Restaurant.find();
            res.json(restaurants);
          } catch (error) {
            res.status(500).json({
              message: "Unable to fetch restaurants",
            });
          }
        });

        app.post("/api/restaurants", async (req, res) => {
          try {
            const restaurant = await Restaurant.create(req.body);
            res.status(201).json(restaurant);
          } catch (error) {
            res.status(400).json({
              message: "Unable to create restaurant",
            });
          }
        });

        const PORT = process.env.PORT || 5000;

        async function startServer() {
          try {
            if (!process.env.MONGO_URI) {
              throw new Error("MONGO_URI is missing from .env");
            }

            await mongoose.connect(process.env.MONGO_URI);

            app.listen(PORT, () => {
              console.log(`Server running on port ${PORT}`);
            });
          } catch (error) {
            console.error("Server startup failed:", error.message);
            process.exit(1);
          }
        }

        startServer();

        Save the file.
